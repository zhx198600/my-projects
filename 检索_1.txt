function testAPI(path, method = 'GET', data = null) {
  return new Promise((resolve, reject) => {
    const options = {
      hostname: 'localhost',
      port: 3001,
      path: path,
      method: method,
      headers: { 'Content-Type': 'application/json' }
    };
    const req = http.request(options, (res) => {
      let body = '';
      res.on('data', (chunk) => body += chunk);
      res.on('end', () => resolve({ status: res.statusCode, body: JSON.parse(body) }));

    if (data) req.write(JSON.stringify(data));

### ✅ 2. Sequelize数据库模型创建

#### 📁 **Document 文档/文件表**
**字段**:
- `title` - 文件标题（必填）
- `description` - 文件描述
- `content` - 文件文本内容（用于搜索）
- `fileName` - 原始文件名
- `filePath` - 文件存储路径
- `fileSize` - 文件大小(字节)
- `fileType` - MIME类型
- `fileExt` - 文件扩展名
- `downloads` - 下载次数（默认0）
- `status` - 发布状态（默认true）
- `tags` - 标签
- `author` - 作者名称
- `categoryId` - 分类外键
